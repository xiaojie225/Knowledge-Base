#### 显示失败提示框
```js
/**
 * 作用：

 * @description 封装 wx.showModal 用于统一展示错误提示信息，并支持回调操作。
 * @param {string} msg - 提示的内容
 * @param {function} [callback] - (可选) 点击确定按钮后的回调函数
 * @returns {void}
 * 例子：this.showError('网络连接失败', () => { wx.navigateBack() });
 */
function showError(msg, callback) {
  wx.showModal({
    title: '温馨提示',
    content: msg,
    showCancel: false,
    success(res) {
      callback && callback();
    }
  });
}
```

#### 小程序主动更新检查
```js

/**
 * 作用：小程序主动更新检查
 * @description 检查小程序是否有新版本，并在下载完成后提示用户重启应用。
 * @returns {void}
 * 例子：this.updateManager();
 */
function updateManager() {
  //canIUse('getUpdateManager')判断当前客户端版本是否支持小程序主动更新
  if (!wx.canIUse('getUpdateManager')) {
    return false;
  }
  //获取更新管理器
  const updateManager = wx.getUpdateManager();
  //onCheckForUpdate检查是否有新版本,如果有新版本,则会触发onCheckForUpdate事件
  updateManager.onCheckForUpdate(res => {
    // 请求完新版本信息的回调
  });
  //onUpdateReady是否下载完成,如果下载完成,则会触发onUpdateReady事件
  updateManager.onUpdateReady(() => {
    wx.showModal({
      title: '更新提示',
      content: '新版本已经准备好，即将重启应用',
      showCancel: false,
      success(res) {
        if (res.confirm) {
          updateManager.applyUpdate()
        }
      }
    });
  });
  //onUpdateFailed是否下载失败,如果下载失败,则会触发onUpdateFailed事件
  updateManager.onUpdateFailed(() => {
    wx.showModal({
      title: '更新提示',
      content: '新版本下载失败',
      showCancel: false
    })
  });
}
```

### **getDistance()** - 计算两点间距离

```js
/**
 * @description 通过经纬度计算两点间的直线距离（使用 Haversine 公式）
 * @param {number} lat1 - 点1的纬度
 * @param {number} lng1 - 点1的经度
 * @param {number} lat2 - 点2的纬度
 * @param {number} lng2 - 点2的经度
 * @returns {string} 返回保留两位小数的距离值，单位为公里
 * @example
 * const distance = getDistance(39.915, 116.404, 31.230, 121.473)
 * console.log(distance) // '1068.84'
 */
function getDistance(lat1, lng1, lat2, lng2) {
  var radLat1 = (lat1 * Math.PI) / 180.0
  var radLat2 = (lat2 * Math.PI) / 180.0
  var a = radLat1 - radLat2
  var b = (lng1 * Math.PI) / 180.0 - (lng2 * Math.PI) / 180.0
  var s =
    2 *
    Math.asin(
      Math.sqrt(
        Math.pow(Math.sin(a / 2), 2) +
          Math.cos(radLat1) *
            Math.cos(radLat2) *
            Math.pow(Math.sin(b / 2), 2)
      )
    )
  s = s * 6378.137 // 地球半径，单位公里
  s = Math.round(s * 10000) / 10000
  s = s.toFixed(2) // 保留两位小数
  return s
}
```

### **getLocationWithAuth()** - 获取地理位置（带权限处理）

```js
/**
 * @description 获取用户当前地理位置（带授权处理）
 * @returns {Promise<{latitude: number, longitude: number}>} 返回包含经纬度的 Promise 对象
 * @example
 * getLocationWithAuth().then(res => {
 *   console.log('纬度:', res.latitude, '经度:', res.longitude)
 * }).catch(err => {
 *   console.error('获取位置失败:', err)
 * })
 */
function getLocationWithAuth() {
  return new Promise((resolve, reject) => {
    wx.authorize({
      scope: 'scope.userLocation',
      success() {
        wx.getLocation({
          type: 'gcj02', // 返回国测局坐标系
          isHighAccuracy: true,
          success(res) {
            resolve({
              latitude: res.latitude,
              longitude: res.longitude
            })
          },
          fail(err) {
            reject(err)
          }
        })
      },
      fail(err) {
        reject(err)
      }
    })
  })
}
```


### **arrayToMapMarkers()** - 将数据列表转换为地图标记


```js
/**
 * @description 将数据列表转换为地图 markers 格式
 * @param {Array} list - 包含经纬度信息的数据列表
 * @param {string} latKey - 纬度字段名，默认为 'latitude'
 * @param {string} lngKey - 经度字段名，默认为 'longitude'
 * @param {string} idKey - id字段名，默认为 'id'
 * @param {string} nameKey - 名称字段名，默认为 'name'
 * @param {Object} styleConfig - 标记样式配置
 * @param {string} styleConfig.iconPath - 图标路径，默认为 '/images/location.png'
 * @param {string} styleConfig.width - 图标宽度，默认为 '50px'
 * @param {string} styleConfig.height - 图标高度，默认为 '50px'
 * @returns {Array} 返回符合微信地图组件 marker 格式的数组
 * @example
 * const markers = arrayToMapMarkers(serviceList, 'lat', 'lon', 'id', 'title', {
 *   iconPath: '/images/marker.png',
 *   width: '40px',
 *   height: '40px'
 * })
 */
function arrayToMapMarkers(list, latKey, lngKey, idKey, nameKey, iconConfig = {}) {
  const {
    iconPath = '/images/location.png',
    width = '50px',
    height = '50px'
  } = iconConfig

  return list.map(item => ({
    id: item[idKey],
    latitude: item[latKey],
    longitude: item[lngKey],
    iconPath: iconPath,
    width: width,
    height: height,
    callout: {
      content: item[nameKey],
      color: '#fff',
      fontSize: 14,
      borderWidth: 1,
      borderRadius: 10,
      borderColor: '#F5F5F5',
      bgColor: '#1BC6A4',
      padding: 5,
      display: 'ALWAYS',
      textAlign: 'center'
    }
  }))
}
```
